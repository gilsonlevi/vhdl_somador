
library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

entity somador1bit is
    Port (
        A    : in  STD_LOGIC;
        B    : in  STD_LOGIC;
        Cin  : in  STD_LOGIC;
        S    : out STD_LOGIC;
        Cout : out STD_LOGIC
    );
end somador1bit;

architecture comportamental of somador1bit is
begin

    S <= A xor B xor Cin;

    Cout <= (A and B) or
            (A and Cin) or
            (B and Cin);

end comportamental;
