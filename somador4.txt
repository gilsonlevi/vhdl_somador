
library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

entity somador4bits is
    Port (
        A   : in  STD_LOGIC_VECTOR(3 downto 0);
        B   : in  STD_LOGIC_VECTOR(3 downto 0);
        Cin : in  STD_LOGIC;
        S   : out STD_LOGIC_VECTOR(3 downto 0);
        Co  : out STD_LOGIC
    );
end somador4bits;

architecture estrutural of somador4bits is

    component somador1bit is
        Port (
            A    : in  STD_LOGIC;
            B    : in  STD_LOGIC;
            Cin  : in  STD_LOGIC;
            S    : out STD_LOGIC;
            Cout : out STD_LOGIC
        );
    end component;

    signal C1, C2, C3 : STD_LOGIC;

begin

    U0: somador1bit
        port map (
            A => A(0),
            B => B(0),
            Cin => Cin,
            S => S(0),
            Cout => C1
        );

    U1: somador1bit
        port map (
            A => A(1),
            B => B(1),
            Cin => C1,
            S => S(1),
            Cout => C2
        );

    U2: somador1bit
        port map (
            A => A(2),
            B => B(2),
            Cin => C2,
            S => S(2),
            Cout => C3
        );

    U3: somador1bit
        port map (
            A => A(3),
            B => B(3),
            Cin => C3,
            S => S(3),
            Cout => Co
        );

end estrutural;
